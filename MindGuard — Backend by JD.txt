from beanie import init_beanie
from app.config import settings
from app.models.user import User
from app.models.guardian import GuardianMapping
from app.models.behaviour_log import BehaviourLog
from app.models.alert import Alert
from app.models.score import DailyScore

client: AsyncIOMotorClient = None

async def connect_db():
    global client
    client = AsyncIOMotorClient(settings.MONGO_URI)
    db = client[settings.MONGO_DB_NAME]
    await init_beanie(
        database=db,
        document_models=[User, GuardianMapping, BehaviourLog, Alert, DailyScore]
    )
    print(f"✅ Connected to MongoDB: {settings.MONGO_DB_NAME}")

async def close_db():
    global client
    if client:
        client.close()
        print("🔌 MongoDB connection closed")

    One document per user per day.

    date: date

from datetime import datetime, timedelta
from typing import Optional
from jose import JWTError, jwt
from passlib.context import CryptContext
from app.config import settings

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

# ── Password ───────────────────────────────────────────────────

def hash_password(plain: str) -> str:
    return pwd_context.hash(plain)

def verify_password(plain: str, hashed: str) -> bool:
    return pwd_context.verify(plain, hashed)

# ── JWT ────────────────────────────────────────────────────────

def create_access_token(data: dict, expires_delta: Optional[timedelta] = None) -> str:
    to_encode = data.copy()
    expire = datetime.utcnow() + (
        expires_delta or timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    )
    to_encode.update({"exp": expire, "type": "access"})
    return jwt.encode(to_encode, settings.JWT_SECRET_KEY, algorithm=settings.JWT_ALGORITHM)

def create_refresh_token(data: dict) -> str:
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(days=settings.REFRESH_TOKEN_EXPIRE_DAYS)
    to_encode.update({"exp": expire, "type": "refresh"})
    return jwt.encode(to_encode, settings.JWT_SECRET_KEY, algorithm=settings.JWT_ALGORITHM)

def decode_token(token: str) -> Optional[dict]:
    try:
        return jwt.decode(token, settings.JWT_SECRET_KEY, algorithms=[settings.JWT_ALGORITHM])
    except JWTError:
        return None

def create_guardian_invite_token(guardian_mapping_id: str) -> str:
    """One-time invite token for guardian to accept the link."""
    data = {"sub": guardian_mapping_id, "type": "guardian_invite"}
    expire = datetime.utcnow() + timedelta(days=7)
    return jwt.encode(
        {**data, "exp": expire}, settings.JWT_SECRET_KEY,
        algorithm=settings.JWT_ALGORITHM
    )

from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from app.core.security import decode_token
from app.models.user import User

bearer_scheme = HTTPBearer()

async def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(bearer_scheme)
) -> User:
    token = credentials.credentials
    payload = decode_token(token)
    if not payload or payload.get("type") != "access":
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired token"
        )
    user = await User.get(payload["sub"])
    if not user or not user.is_active:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="User not found")
    return user

async def require_user_role(current_user: User = Depends(get_current_user)) -> User:
    if current_user.role not in ("user", "admin"):
        raise HTTPException(status_code=403, detail="User role required")
    return current_user

async def require_guardian_role(current_user: User = Depends(get_current_user)) -> User:
    if current_user.role not in ("guardian", "admin"):
        raise HTTPException(status_code=403, detail="Guardian role required")
    return current_user

async def require_admin(current_user: User = Depends(get_current_user)) -> User:
    if current_user.role != "admin":
        raise HTTPException(status_code=403, detail="Admin role required")
    return current_user

import socketio
from app.core.security import decode_token

# Async Socket.IO server (ASGI-compatible)
sio = socketio.AsyncServer(
    async_mode="asgi",
    cors_allowed_origins="*",   # tighten in production
    logger=False,
    engineio_logger=False
)

# ── Connection lifecycle ───────────────────────────────────────

@sio.event
async def connect(sid, environ, auth):
    """Authenticate socket connection via JWT."""
    token = (auth or {}).get("token")
    if not token:
        return False  # reject
    payload = decode_token(token)
    if not payload:
        return False
    user_id = payload.get("sub")
    if not user_id:
        return False
    # Join a private room keyed by user_id
    await sio.enter_room(sid, f"user_{user_id}")
    await sio.save_session(sid, {"user_id": user_id, "role": payload.get("role","user")})
    print(f"[Socket] {sid} connected — user {user_id}")

@sio.event
async def disconnect(sid):
    print(f"[Socket] {sid} disconnected")

# ── Emitter helpers ───────────────────────────────────────────

async def emit_to_user(user_id: str, event: str, data: dict):
    """Push an event to all sockets belonging to a user."""
    await sio.emit(event, data, room=f"user_{user_id}")

async def emit_score_update(user_id: str, score_data: dict):
    await emit_to_user(user_id, "score_update", score_data)

async def emit_alert(user_id: str, alert_data: dict):
    await emit_to_user(user_id, "new_alert", alert_data)

async def emit_guardian_notification(user_id: str, notif_data: dict):
    await emit_to_user(user_id, "guardian_notified", notif_data)

"""
SCORING ENGINE
━━━━━━━━━━━━━━
Base score: 100
Deductions and bonuses are applied by category.

Deductions:
  Late-night usage  → –3 pts per 30-min block after midnight (max –12)
  Erratic typing    → –3 pts if avg backspace ratio > 0.25 AND wpm > 70
  Doomscrolling     → –2 pts per 30 min of news after 40 min/day (max –6)
  High screen time  → –1 pt per 30 min above personal 30-day avg (max –8)
  Compulsive opens  → –2 pts per app exceeding 20 opens/day (max –6)
  Work-hour avoid   → –3 pts if >40% of work-hour usage is entertainment

Bonuses:
  Wellness app used → +5 pts (Breathwrk, Headspace, Calm, etc.)
  MindGuard use     → +3 pts
  Screen time < avg → +5 pts if >15% below 30-day avg
  Low app opens     → +2 pts if <30 opens total
  Positive mood     → +3 pts if self_mood >= 4
  Creative app use  → +2 pts (Camera, journalling apps etc.)
"""

from datetime import date, datetime, timedelta
from typing import Optional, List
from app.models.behaviour_log import BehaviourLog
from app.models.score import DailyScore, ScoreComponent
from app.models.user import User

WELLNESS_APPS = {"breathwrk", "headspace", "calm", "mindguard", "medito", "insight timer"}
NEWS_APPS = {"news", "inshorts", "google news", "bbc", "cnn", "ndtv", "reuters"}
ENTERTAINMENT_APPS = {"instagram", "tiktok", "youtube", "netflix", "spotify", "reels"}
CREATIVE_APPS = {"camera", "vsco", "notion", "day one", "reflectly"}

STATUS_MAP = [
    (80, "flourishing"),
    (65, "stable"),
    (45, "moderate"),
    (0,  "concerning"),
]

def _status(score: int) -> str:
    for threshold, label in STATUS_MAP:
        if score >= threshold:
            return label
    return "concerning"

async def _avg_screen_time(user_id: str, days: int = 30) -> int:
    """Return average daily screen time in seconds over past N days."""
    since = date.today() - timedelta(days=days)
    logs = await BehaviourLog.find(
        BehaviourLog.user_id == user_id,
        BehaviourLog.date >= since,
        BehaviourLog.date < date.today()
    ).to_list()
    if not logs:
        return 5 * 3600  # default: 5 hours
    return int(sum(l.total_screen_time_seconds for l in logs) / len(logs))

async def calculate_daily_score(log: BehaviourLog, user: User) -> DailyScore:
    components: List[ScoreComponent] = []
    score = 100

    def deduct(pts: int, label: str, category: str):
        nonlocal score
        score = max(0, score - abs(pts))
        components.append(ScoreComponent(label=label, points=-abs(pts), category=category))

    def bonus(pts: int, label: str, category: str):
        nonlocal score
        score = min(100, score + abs(pts))
        components.append(ScoreComponent(label=label, points=abs(pts), category=category))

    # ── DEDUCTIONS ────────────────────────────────────────────

    # 1. Late-night usage
    if log.late_night_minutes > 0:
        blocks = min(int(log.late_night_minutes / 30), 4)
        deduct(blocks * 3, f"Late-night usage ({log.late_night_minutes} min after midnight)", "late_night")

    # 2. Erratic typing
    if log.avg_backspace_ratio and log.avg_wpm:
        if log.avg_backspace_ratio > 0.25 and log.avg_wpm > 70:
            deduct(3, f"Erratic typing detected (backspace ratio: {log.avg_backspace_ratio:.0%})", "typing")

    # 3. Doomscrolling (news apps)
    news_seconds = sum(
        a.duration_seconds for a in log.app_usage
        if any(n in a.app_name.lower() for n in NEWS_APPS)
    )
    if news_seconds > 40 * 60:
        extra_blocks = min(int((news_seconds - 40 * 60) / (30 * 60)), 3)
        deduct(extra_blocks * 2, f"Extended news reading ({news_seconds//60} min)", "doomscrolling")

    # 4. High screen time vs personal average
    avg_screen = await _avg_screen_time(str(log.user_id))
    excess = log.total_screen_time_seconds - avg_screen
    if excess > 0:
        excess_blocks = min(int(excess / (30 * 60)), 8)
        deduct(excess_blocks, f"Screen time {excess//60} min above your average", "screen_time")

    # 5. Compulsive app opens
    for app in log.app_usage:
        if app.open_count > 20:
            deduct(2, f"{app.app_name} opened {app.open_count}× (compulsive threshold)", "compulsive")

    # 6. Work-hour avoidance (basic heuristic using app categories)
    work_start = int(user.work_hours_start.split(":")[0])
    work_end = int(user.work_hours_end.split(":")[0])
    work_hours_usage = sum(log.usage_by_hour[work_start:work_end])
    entertain_during_work = sum(
        a.duration_seconds // 60 for a in log.app_usage
        if a.category in ("social", "entertainment", "gaming")
    )
    if work_hours_usage > 0 and entertain_during_work / max(work_hours_usage, 1) > 0.40:
        deduct(3, "Significant entertainment use during work hours", "work_avoidance")

    # ── BONUSES ───────────────────────────────────────────────

    # 1. Wellness apps
    for app in log.app_usage:
        if any(w in app.app_name.lower() for w in WELLNESS_APPS) and app.duration_seconds > 300:
            bonus(5, f"Wellness app used: {app.app_name} ({app.duration_seconds//60} min)", "wellness")
            break

    # 2. MindGuard self-check
    mindguard_use = any("mindguard" in a.app_name.lower() for a in log.app_usage)
    if mindguard_use:
        bonus(3, "MindGuard check-in completed", "wellness")

    # 3. Low screen time
    if log.total_screen_time_seconds < avg_screen * 0.85:
        bonus(5, f"Screen time {int((1 - log.total_screen_time_seconds/avg_screen)*100)}% below your average", "screen_time")

    # 4. Low total app opens
    total_opens = sum(a.open_count for a in log.app_usage)
    if total_opens < 30:
        bonus(2, f"Low app-open count ({total_opens} opens today)", "mindful")

    # 5. Positive mood self-report
    if log.self_mood and log.self_mood >= 4:
        bonus(3, f"Positive self-reported mood ({log.self_mood}/5)", "mood")

    # 6. Creative apps used
    for app in log.app_usage:
        if any(c in app.app_name.lower() for c in CREATIVE_APPS):
            bonus(2, f"Creative app used: {app.app_name}", "creative")
            break

    summary = _build_summary(score, components)
    daily = DailyScore(
        user_id=str(log.user_id),
        date=log.date,
        score=score,
        status=_status(score),
        components=components,
        summary=summary,
        calculated_at=datetime.utcnow()
    )
    # Upsert
    existing = await DailyScore.find_one(
        DailyScore.user_id == str(log.user_id),
        DailyScore.date == log.date
    )
    if existing:
        existing.score = score
        existing.status = daily.status
        existing.components = components
        existing.summary = summary
        existing.calculated_at = daily.calculated_at
        await existing.save()
        return existing
    await daily.insert()
    return daily

def _build_summary(score: int, components: List[ScoreComponent]) -> str:
    positives = [c.label for c in components if c.points > 0]
    negatives = [c.label for c in components if c.points < 0]
    if score >= 80:
        return "A great day — your patterns reflect a balanced, mindful digital life."
    elif score >= 65:
        return f"A steady day. {negatives[0] if negatives else 'Minor flags'} pulled the score slightly. {positives[0] if positives else ''}"
    elif score >= 45:
        return f"Some patterns to watch today. Key concern: {negatives[0] if negatives else 'multiple flags'}."
    else:
        return "A challenging day digitally. Your guardian has been gently informed. You're not alone."

async def recalculate_today(user_id: str, mood_override: Optional[int] = None) -> DailyScore:
    today = date.today()
    log = await BehaviourLog.find_one(
        BehaviourLog.user_id == user_id,
        BehaviourLog.date == today
    )
    if not log:
        # Return minimal score if no data yet
        return DailyScore(user_id=user_id, date=today, score=100, status="flourishing")
    if mood_override:
        log.self_mood = mood_override
        await log.save()
    user = await User.get(user_id)
    return await calculate_daily_score(log, user)

async def evaluate_alerts(user_id: str, today_score: DailyScore):
    """Run after every score calculation. Checks thresholds and fires alerts."""

    # ── Red alert: immediate single-day trigger ────────────────
    if today_score.score <= settings.RED_ALERT_SCORE:
        await _fire_alert(
            user_id, "red",
            "🔴 Urgent: Very Low Wellness Score",
            f"JD's wellness score has dropped to {today_score.score}/100 today. "
            f"This has triggered an urgent notification to their guardians."
        )
        return

    # ── Yellow alert: N consecutive days below threshold ────────
    since = date.today() - timedelta(days=settings.YELLOW_CONSECUTIVE_DAYS - 1)
    recent_scores = await DailyScore.find(
        DailyScore.user_id == user_id,
        DailyScore.date >= since,
        DailyScore.date <= date.today()
    ).to_list()

    if len(recent_scores) >= settings.YELLOW_CONSECUTIVE_DAYS:
        all_low = all(s.score <= settings.YELLOW_ALERT_SCORE for s in recent_scores)
        if all_low:
            await _fire_alert(
                user_id, "yellow",
                "⚠️ Wellness Score Consistently Low",
                f"JD's score has been at or below {settings.YELLOW_ALERT_SCORE} for "
                f"{settings.YELLOW_CONSECUTIVE_DAYS} consecutive days. "
                f"A gentle summary has been sent to their guardians."
            )

async def _fire_alert(user_id: str, level: str, title: str, message: str):
    # Avoid duplicate unresolved alerts of same level on same day
    today = date.today()
    existing = await Alert.find_one(
        Alert.user_id == user_id,
        Alert.level == level,
        Alert.is_resolved == False
    )
    if existing:
        return  # already active

    alert = Alert(user_id=user_id, level=level, title=title, message=message)

    # Notify relevant guardians
    guardians: list[GuardianMapping] = await GuardianMapping.find(
        GuardianMapping.user_id == user_id,
        GuardianMapping.status == "active"
    ).to_list()

    notified_ids = []
    for g in guardians:
        should_notify = (
            (level == "red" and g.red_alerts) or
            (level == "yellow" and g.yellow_alerts) or
            level == "info"
        )
        if should_notify:
            await send_guardian_email(g, level, title, message)
            notified_ids.append(str(g.id))

    alert.guardian_ids_notified = notified_ids
    await alert.insert()

    # Real-time push to user's dashboard
    await emit_alert(user_id, {
        "id": str(alert.id),
        "level": level,
        "title": title,
        "message": message,
        "triggered_at": alert.triggered_at.isoformat()
    })

    # Notify guardian's own dashboard if they have an account
    for g in guardians:
        if g.guardian_user_id:
            await emit_guardian_notification(g.guardian_user_id, {
                "user_id": user_id,
                "level": level,
                "title": title,
                "message": message
            })

async def get_daily_report(user_id: str, report_date: date) -> dict:
    score = await DailyScore.find_one(
        DailyScore.user_id == user_id,
        DailyScore.date == report_date
    )
    log = await BehaviourLog.find_one(
        BehaviourLog.user_id == user_id,

    )
    return {

        "score": score.score if score else None,

        "components": [c.dict() for c in score.components] if score else [],
        "screen_time_seconds": log.total_screen_time_seconds if log else 0,
        "app_usage": [a.dict() for a in log.app_usage] if log else [],
        "usage_by_hour": log.usage_by_hour if log else [0]*24,
        "late_night_minutes": log.late_night_minutes if log else 0,
        "self_mood": log.self_mood if log else None,
    }

async def get_weekly_report(user_id: str, week_start: date) -> dict:
    week_end = week_start + timedelta(days=6)
    scores = await DailyScore.find(
        DailyScore.user_id == user_id,
        DailyScore.date >= week_start,
        DailyScore.date <= week_end
    ).sort("+date").to_list()

    daily_scores = [{"date": str(s.date), "score": s.score, "status": s.status} for s in scores]
    avg = round(sum(s.score for s in scores) / len(scores), 1) if scores else None
    best = max(scores, key=lambda s: s.score) if scores else None
    worst = min(scores, key=lambda s: s.score) if scores else None

    return {
        "week_start": str(week_start),
        "week_end": str(week_end),
        "average_score": avg,
        "daily_scores": daily_scores,
        "best_day": {"date": str(best.date), "score": best.score} if best else None,
        "worst_day": {"date": str(worst.date), "score": worst.score} if worst else None,
        "days_above_75": sum(1 for s in scores if s.score >= 75),
        "days_below_45": sum(1 for s in scores if s.score < 45),
    }

async def get_monthly_report(user_id: str, year: int, month: int) -> dict:
    from calendar import monthrange
    _, days_in_month = monthrange(year, month)
    month_start = date(year, month, 1)
    month_end = date(year, month, days_in_month)
    scores = await DailyScore.find(
        DailyScore.user_id == user_id,
        DailyScore.date >= month_start,
        DailyScore.date <= month_end
    ).sort("+date").to_list()

    avg = round(sum(s.score for s in scores) / len(scores), 1) if scores else None
    return {
        "year": year, "month": month,
        "average_score": avg,
        "total_days_tracked": len(scores),
        "days_above_75": sum(1 for s in scores if s.score >= 75),
        "days_below_45": sum(1 for s in scores if s.score < 45),
        "daily_scores": [{"date": str(s.date), "score": s.score} for s in scores],
    }

async def get_yearly_report(user_id: str, year: int) -> dict:
    year_start = date(year, 1, 1)
    year_end = date(year, 12, 31)
    scores = await DailyScore.find(
        DailyScore.user_id == user_id,
        DailyScore.date >= year_start,
        DailyScore.date <= year_end
    ).sort("+date").to_list()

    # Group by month
    monthly: dict = {}
    for s in scores:
        key = s.date.month
        monthly.setdefault(key, []).append(s.score)
    monthly_avgs = [
        {"month": m, "average": round(sum(v)/len(v), 1), "days": len(v)}
        for m, v in sorted(monthly.items())
    ]
    overall_avg = round(sum(s.score for s in scores) / len(scores), 1) if scores else None
    return {
        "year": year,
        "overall_average": overall_avg,
        "monthly_averages": monthly_avgs,
        "total_days_tracked": len(scores),
    }

async def dispatch_weekly_reports(user_id: str):
    """Called by a scheduler every Sunday morning."""
    today = date.today()
    week_start = today - timedelta(days=today.weekday() + 1)  # last Monday
    report = await get_weekly_report(user_id, week_start)
    guardians = await GuardianMapping.find(
        GuardianMapping.user_id == user_id,
        GuardianMapping.status == "active",
        GuardianMapping.receives_weekly == True
    ).to_list()
    for g in guardians:
        await send_weekly_report_email(g, report)

async def _send_email(to_email: str, subject: str, html_body: str):
    msg = MIMEMultipart("alternative")
    msg["Subject"] = subject
    msg["From"] = settings.EMAIL_FROM
    msg["To"] = to_email
    msg.attach(MIMEText(html_body, "html"))
    try:
        await aiosmtplib.send(
            msg,
            hostname=settings.SMTP_HOST,
            port=settings.SMTP_PORT,
            username=settings.SMTP_USER,
            password=settings.SMTP_PASSWORD,
            start_tls=True
        )
        print(f"[Email] Sent '{subject}' to {to_email}")
    except Exception as e:
        print(f"[Email] Failed to send to {to_email}: {e}")

"""
AES-256-GCM field-level encryption for sensitive stored fields
(self_notes, typing_events detail, etc.)
"""
import base64
import os
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
from app.config import settings

def _get_key() -> bytes:
    return base64.b64decode(settings.FIELD_ENCRYPTION_KEY)

def encrypt(plaintext: str) -> str:
    key = _get_key()
    aesgcm = AESGCM(key)
    nonce = os.urandom(12)
    ct = aesgcm.encrypt(nonce, plaintext.encode(), None)
    return base64.b64encode(nonce + ct).decode()

def decrypt(ciphertext: str) -> str:
    key = _get_key()
    aesgcm = AESGCM(key)
    raw = base64.b64decode(ciphertext)
    nonce, ct = raw[:12], raw[12:]
    return aesgcm.decrypt(nonce, ct, None).decode()

from fastapi import APIRouter, HTTPException, status

    hash_password, verify_password,
    create_access_token, create_refresh_token, decode_token
)
from datetime import datetime
from pydantic import BaseModel

router = APIRouter(prefix="/auth", tags=["Authentication"])

class TokenResponse(BaseModel):
    access_token: str
    refresh_token: str
    token_type: str = "bearer"
    user: UserOut

class RefreshRequest(BaseModel):
    refresh_token: str

@router.post("/register", response_model=TokenResponse, status_code=201)
async def register(data: UserRegister):
    if await User.find_one(User.email == data.email):
        raise HTTPException(status_code=400, detail="Email already registered")
    user = User(
        email=data.email,
        hashed_password=hash_password(data.password),
        full_name=data.full_name,
        phone=data.phone,
        date_of_birth=data.date_of_birth,
    )
    await user.insert()
    return _make_token_response(user)

@router.post("/login", response_model=TokenResponse)
async def login(data: UserLogin):
    user = await User.find_one(User.email == data.email)
    if not user or not verify_password(data.password, user.hashed_password):
        raise HTTPException(status_code=401, detail="Invalid credentials")
    if not user.is_active:
        raise HTTPException(status_code=403, detail="Account disabled")
    return _make_token_response(user)

@router.post("/refresh", response_model=TokenResponse)
async def refresh_token(data: RefreshRequest):
    payload = decode_token(data.refresh_token)
    if not payload or payload.get("type") != "refresh":
        raise HTTPException(status_code=401, detail="Invalid refresh token")
    user = await User.get(payload["sub"])
    if not user:
        raise HTTPException(status_code=401, detail="User not found")
    return _make_token_response(user)

@router.post("/logout")
async def logout():
    # Stateless JWT — client simply discards tokens
    # In production: add token to a Redis blocklist here
    return {"detail": "Logged out successfully"}

def _make_token_response(user: User) -> dict:
    uid = str(user.id)
    return {
        "access_token": create_access_token({"sub": uid, "role": user.role}),
        "refresh_token": create_refresh_token({"sub": uid}),
        "user": UserOut(
            id=uid, email=user.email, full_name=user.full_name,
            phone=user.phone, role=user.role, is_active=user.is_active,
            created_at=user.created_at
        )
    }

@router.post("/ingest")
async def ingest_behaviour(
    data: BehaviourIngestRequest,
    current_user: User = Depends(require_user_role)
):
    """
    Primary data ingest endpoint.
    Called by the mobile SDK / browser agent when syncing usage data.
    Automatically re-calculates the score and fires alert checks.
    """
    uid = str(current_user.id)

    # Compute aggregate typing stats
    avg_wpm, avg_backspace = None, None
    if data.typing_events:
        avg_wpm = statistics.mean(e.wpm for e in data.typing_events)
        avg_backspace = statistics.mean(e.backspace_ratio for e in data.typing_events)

    # Encrypt self_notes if present
    encrypted_notes = encrypt(data.self_notes) if data.self_notes else None

    # Upsert log
    log = await BehaviourLog.find_one(
        BehaviourLog.user_id == uid,
        BehaviourLog.date == data.date
    )
    if log:
        log.app_usage = data.app_usage
        log.total_screen_time_seconds = data.total_screen_time_seconds
        log.typing_events = data.typing_events
        log.avg_wpm = avg_wpm
        log.avg_backspace_ratio = avg_backspace
        log.late_night_minutes = data.late_night_minutes
        log.usage_by_hour = data.usage_by_hour
        if data.self_mood:
            log.self_mood = data.self_mood
        if encrypted_notes:
            log.self_notes = encrypted_notes
        log.updated_at = datetime.utcnow()
        await log.save()
    else:
        log = BehaviourLog(
            user_id=uid, date=data.date,
            app_usage=data.app_usage,
            total_screen_time_seconds=data.total_screen_time_seconds,
            typing_events=data.typing_events,
            avg_wpm=avg_wpm, avg_backspace_ratio=avg_backspace,
            late_night_minutes=data.late_night_minutes,
            usage_by_hour=data.usage_by_hour,
            self_mood=data.self_mood,
            self_notes=encrypted_notes
        )
        await log.insert()

    # Score + alert pipeline
    score = await calculate_daily_score(log, current_user)
    await evaluate_alerts(uid, score)

    # Real-time push
    await emit_score_update(uid, {"score": score.score, "status": score.status, "date": str(data.date)})

    return {"score": score.score, "status": score.status, "components": [c.dict() for c in score.components]}

from fastapi import APIRouter, Depends
from app.models.user import User, UserOut, UserUpdate
from app.core.dependencies import require_user_role
from datetime import datetime

router = APIRouter(prefix="/users", tags=["Users"])

@router.get("/me", response_model=UserOut)
async def get_me(current_user: User = Depends(require_user_role)):
    return UserOut(
        id=str(current_user.id), email=current_user.email,
        full_name=current_user.full_name, phone=current_user.phone,
        role=current_user.role, is_active=current_user.is_active,
        created_at=current_user.created_at
    )

@router.patch("/me")
async def update_me(data: UserUpdate, current_user: User = Depends(require_user_role)):
    update_data = data.dict(exclude_none=True)
    for k, v in update_data.items():
        setattr(current_user, k, v)
    current_user.updated_at = datetime.utcnow()
    await current_user.save()
    return {"detail": "Profile updated"}

@router.delete("/me")
async def delete_account(current_user: User = Depends(require_user_role)):
    """Soft-delete: deactivate account and wipe behavioural logs."""
    from app.models.behaviour_log import BehaviourLog
    from app.models.score import DailyScore
    current_user.is_active = False
    await current_user.save()
    await BehaviourLog.find(BehaviourLog.user_id == str(current_user.id)).delete()
    await DailyScore.find(DailyScore.user_id == str(current_user.id)).delete()
    return {"detail": "Account deactivated and data deleted"}
